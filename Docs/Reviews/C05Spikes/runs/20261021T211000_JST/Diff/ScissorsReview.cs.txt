        report.cases.Add(record);
        Check(attached?record.attachedAfter==!cut:!record.attachedAfter,"Native trigger "+name);
        Check(record.remainingAfter==record.remainingBefore,"No duplicate rope/refund amount "+name);
        Check(record.snipsAfter-record.snipsBefore==(cut?1:0),"Single/zero reaction "+name);
    }
