using System;
using System.IO;
using System.Reflection;
using System.Security.Cryptography;
using UnityEngine;

// ONE isolated Front hazard. Existing RopeSpikeHazard still owns cut/refund/contact effects.
[DefaultExecutionOrder(-200)]
public sealed class CraftScissorsMotion : MonoBehaviour
{
    public const string TargetName="Main Section 2 Front Spike";
    public const float FloorY=-8.34f,PPU=500f;
    public const float CloseSeconds=.18f,HoldSeconds=.08f,ReturnSeconds=.48f;
    public static string AssetHash;
    // This candidate changes placement only. Everything uses the same lower support reference.
    public static float ReviewSupportY=FloorY;
    public float SupportY { get; private set; }
    public PolygonCollider2D blades;
    public BoxCollider2D oldBox;
    public Transform left,right,screw,root;
    public int snips;
    public float closure;
    public Vector3 leftSupport,rightSupport;
    public bool diagnosticHold;
    public float diagnosticClosure;
    public string Phase=>cycle<0?"open idle":cycle<CloseSeconds?"closing":cycle<CloseSeconds+HoldSeconds?"closed":"reopening";
    float idle,cycle=-1,lastContact=.4f,cycleStart;
    Transform shadowL,shadowR;SpriteRenderer oldFace;
    HimoHito.ToySpikeVisual oldVisual;
    FieldInfo contactField;
    Geometry geometry;
    static Texture2D atlas;
    [Serializable]public class Geometry{public Vector2[] leftHandle,rightHandle,leftBlade,rightBlade;}
    [Serializable]public class Pose{public string phase;public float closure,rootZ,tiltX,tiltY,pivotDistance,handleGap;public Vector3 hinge,leftSupport,rightSupport;public Vector3[] leftBlade,rightBlade;}
    public Pose Record()=>new Pose{phase=Phase,closure=closure,rootZ=root.eulerAngles.z,tiltX=root.eulerAngles.x,tiltY=root.eulerAngles.y,hinge=root.position,leftSupport=leftSupport,rightSupport=rightSupport,handleGap=Mathf.Abs(leftSupport.y-rightSupport.y),pivotDistance=Mathf.Max(Vector3.Distance(left.position,right.position),Vector3.Distance(left.position,screw.position)),leftBlade=WorldBlade(0),rightBlade=WorldBlade(1)};
    void Awake()
    {
        if(name!=TargetName)throw new InvalidOperationException("C05 scope guard: Front only");
        string dir=Path.Combine(Path.GetDirectoryName(Application.dataPath),"ReviewAssets");
        byte[] bytes=File.ReadAllBytes(Path.Combine(dir,"HimoHitoWarmScissorsParts-v3.png"));
        using(var sha=SHA256.Create())AssetHash=BitConverter.ToString(sha.ComputeHash(bytes)).Replace("-","").ToLowerInvariant();
        if(AssetHash!="b5558bcd4b40b7762201184ec0324758803aaaede56cc3102dbd169670c57c1d")throw new Exception("Approved art mismatch");
        if(!atlas){atlas=new Texture2D(2,2,TextureFormat.RGBA32,true){filterMode=FilterMode.Trilinear,wrapMode=TextureWrapMode.Clamp};atlas.LoadImage(bytes);}
        geometry=JsonUtility.FromJson<Geometry>(File.ReadAllText(Path.Combine(dir,"BladeGeometry.json")));
        oldVisual=GetComponent<HimoHito.ToySpikeVisual>();
        contactField=typeof(HimoHito.ToySpikeVisual).GetField("contactElapsed",BindingFlags.Instance|BindingFlags.NonPublic);
        oldFace=transform.Find("Wooden Toy Spike Face").GetComponent<SpriteRenderer>();
        var space=new GameObject("C05 unscaled candidate").transform;space.SetParent(transform,false);
        space.localScale=new Vector3(1/transform.lossyScale.x,1/transform.lossyScale.y,1/transform.lossyScale.z);
        root=new GameObject("Common moving pivot; upright front").transform;root.SetParent(space,false);
        root.position=new Vector3(transform.position.x,0,transform.position.z);root.localRotation=Quaternion.Euler(0,0,180);
        root.localScale=Vector3.one;
        SupportY=ReviewSupportY;
        var sl=Sprite.Create(atlas,new Rect(230,0,390,1024),new Vector2(317f/390,541f/1024),PPU,0,SpriteMeshType.FullRect);
        var sr=Sprite.Create(atlas,new Rect(830,0,400,1024),new Vector2(74f/400,541f/1024),PPU,0,SpriteMeshType.FullRect);
        var ss=Sprite.Create(atlas,new Rect(1273,479,111,112),new Vector2(55f/111,56f/112),PPU,0,SpriteMeshType.FullRect);
        shadowL=Part(sl,"Left shadow",2,new Color(.055f,.025f,.015f,.27f));shadowR=Part(sr,"Right shadow",2,new Color(.055f,.025f,.015f,.21f));
        shadowL.localPosition=shadowR.localPosition=Quaternion.Inverse(root.localRotation)*new Vector3(.012f,-.035f,0);
        left=Part(sl,"Left rigid handle through blade",3,Color.white);right=Part(sr,"Right rigid handle through blade",4,Color.white);screw=Part(ss,"Original shared screw",5,Color.white);
        oldBox=GetComponent<BoxCollider2D>();oldBox.enabled=false;
        blades=gameObject.AddComponent<PolygonCollider2D>();blades.isTrigger=true;blades.pathCount=2;
        Apply(0);
        Debug.Log("C05_MOTION_READY one Front only; original cut/refund; PNG="+AssetHash);
    }
    Transform Part(Sprite sprite,string name,int order,Color color){var g=new GameObject(name);g.transform.SetParent(root,false);var s=g.AddComponent<SpriteRenderer>();s.sprite=sprite;s.color=color;s.sortingLayerID=oldFace.sortingLayerID;s.sortingOrder=order;return g.transform;}
    static float Smooth(float t)=>Mathf.SmoothStep(0,1,Mathf.Clamp01(t));
    void FixedUpdate()
    {
        if(diagnosticHold){Apply(diagnosticClosure);return;}
        if(cycle>=0){cycle+=Time.fixedDeltaTime;if(cycle<CloseSeconds)Apply(Mathf.Lerp(cycleStart,1,Smooth(cycle/CloseSeconds)));else if(cycle<CloseSeconds+HoldSeconds)Apply(1);else if(cycle<CloseSeconds+HoldSeconds+ReturnSeconds)Apply(1-Smooth((cycle-CloseSeconds-HoldSeconds)/ReturnSeconds));else{cycle=-1;idle=0;Apply(0);}}
        else{idle+=Time.fixedDeltaTime;Apply(.035f*(1-Mathf.Cos(idle*1.35f))*.5f);}
    }
    void LateUpdate()
    {
        if(oldFace)oldFace.enabled=false;
        float contact=(float)contactField.GetValue(oldVisual);
        // Observe the existing native contact effect; never invoke Cut/Detach/refund ourselves.
        if(Time.timeScale>0&&contact<lastContact-.01f){snips++;if(cycle<0){cycleStart=closure;cycle=0;}}
        lastContact=contact;
    }
    public void Apply(float value)
    {
        closure=Mathf.Clamp01(value);
        left.localRotation=Quaternion.Euler(0,0,Mathf.Lerp(16,-3.4f,closure));right.localRotation=Quaternion.Euler(0,0,Mathf.Lerp(-16,3.8f,closure));
        shadowL.localRotation=left.localRotation;shadowR.localRotation=right.localRotation;
        leftSupport=Lowest(left,geometry.leftHandle);rightSupport=Lowest(right,geometry.rightHandle);
        root.position+=Vector3.up*(SupportY-Mathf.Min(leftSupport.y,rightSupport.y));
        leftSupport=Lowest(left,geometry.leftHandle);rightSupport=Lowest(right,geometry.rightHandle);
        for(int side=0;side<2;side++){var w=WorldBlade(side);var local=new Vector2[w.Length];for(int i=0;i<w.Length;i++)local[i]=transform.InverseTransformPoint(w[i]);blades.SetPath(side,local);}
    }
    static Vector3 Lowest(Transform t,Vector2[] hull){var best=new Vector3(0,float.PositiveInfinity,0);foreach(var p in hull){var w=t.TransformPoint(p);if(w.y<best.y)best=w;}return best;}
    public Vector3[] WorldBlade(int side){var src=side==0?geometry.leftBlade:geometry.rightBlade;var t=side==0?left:right;var dst=new Vector3[src.Length];for(int i=0;i<src.Length;i++)dst[i]=t.TransformPoint(src[i]);return dst;}
    // Review fixture only uses these to inspect poses. Manual launch never calls them.
    public void DiagnosticPose(float c){diagnosticHold=true;diagnosticClosure=c;Apply(c);}
    public void DiagnosticRelease(){diagnosticHold=false;cycle=-1;idle=0;}
}
