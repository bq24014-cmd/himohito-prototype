    float CurveAt(LineRenderer line,float x) {
        for(int i=0;i<line.positionCount-1;i++) {
            Vector3 a=line.GetPosition(i),b=line.GetPosition(i+1);
            if(!line.useWorldSpace) { a=line.transform.TransformPoint(a); b=line.transform.TransformPoint(b); }
            if(x>=Mathf.Min(a.x,b.x) && x<=Mathf.Max(a.x,b.x)) return Mathf.Lerp(a.y,b.y,(x-a.x)/(b.x-a.x));
        }
        return -999;
    }
    readonly Dictionary<Sprite,Vector2> soleCache=new Dictionary<Sprite,Vector2>();
    Vector2 Sole(Sprite sprite) {
        if(soleCache.TryGetValue(sprite,out var cached)) return cached;
        Texture2D source=sprite.texture; var rt=RenderTexture.GetTemporary(source.width,source.height,0,RenderTextureFormat.ARGB32);
        var previous=RenderTexture.active; Graphics.Blit(source,rt); RenderTexture.active=rt;
        var readable=new Texture2D(source.width,source.height,TextureFormat.RGBA32,false);
        readable.ReadPixels(new Rect(0,0,source.width,source.height),0,0); readable.Apply();
        RenderTexture.active=previous; RenderTexture.ReleaseTemporary(rt); var pixels=readable.GetPixels32(); Destroy(readable);
        Rect r=sprite.rect; int minY=(int)r.height, sumX=0,count=0;
        for(int y=0;y<(int)r.height;y++) {
            for(int x=(int)(r.width*.2f);x<(int)(r.width*.8f);x++)
                if(pixels[((int)r.y+y)*source.width+(int)r.x+x].a>80) { minY=y; sumX+=x; count++; }
            if(count>0) break;
        }
        cached=new Vector2(((count>0?sumX/(float)count+.5f:sprite.pivot.x)-sprite.pivot.x)/sprite.pixelsPerUnit,
            ((count>0?minY+.5f:sprite.pivot.y)-sprite.pivot.y)/sprite.pixelsPerUnit);
        soleCache[sprite]=cached; return cached;
    }
    State Sample() {
        var renderer=lower && lower.enabled?lower:art; Vector2 sole=Sole(renderer.sprite); if(renderer.flipX) sole.x=-sole.x;
        Vector3 foot=renderer.transform.TransformPoint(sole); var edge=bridge.GetComponent<EdgeCollider2D>();
        if(!geometries.Any(g=>g.start==bridge.Start && g.end==bridge.End && Mathf.Abs(g.length-bridge.RopeLength)<.001f))
            geometries.Add(new Geometry{start=bridge.Start,end=bridge.End,length=bridge.RopeLength,radius=edge.edgeRadius,points=edge.points});
        var line=bridge.GetComponent<LineRenderer>(); float colliderY=-999;
        foreach(var hit in Physics2D.RaycastAll(new Vector2(body.position.x,body.position.y+2),Vector2.down,20))
            if(hit.collider==edge) { colliderY=hit.point.y; break; }
        var separation=box.Distance(edge); float lineY=CurveAt(line,body.position.x);
        float normalized=Mathf.InverseLerp(bridge.Start.x,bridge.End.x,body.position.x);
        float width=line.widthMultiplier*line.widthCurve.Evaluate(normalized);
        bridge.TryGetSurfaceTangent(new Vector2(body.position.x,lineY),out Vector2 tangent);
        // Diagnostic approximation of the rendered upper edge. Line width is
        // widthCurve * widthMultiplier, not widthMultiplier alone.
        float visibleTop=lineY+width*.5f/Mathf.Max(.2f,Mathf.Abs(tangent.x));
        var row=new State { scenario=caseName,phase=phase,scene=gameObject.scene.name,section=run?run.CurrentTutorialSection:mainRun.CurrentSection,bridge=bridgeKind,
            frame=Time.frameCount,time=Time.time,remaining=resource.CurrentLength,scale=(float)Field(visual,"currentBaseScale"),
            position=body.position,velocity=body.linearVelocity,colliderBottom=box.bounds.min.y,bodyWidth=box.bounds.size.x,bodyHeight=box.bounds.size.y,
            bridgeStart=bridge.Start,bridgeEnd=bridge.End,bridgeColliderAtCenter=colliderY,bridgeLineAtCenter=lineY,bridgeVisualTopAtCenter=visibleTop,bridgeWidth=width,
            opaqueSoleY=foot.y,soleGap=foot.y-visibleTop,closestBridgePointX=separation.pointB.x,closestBridgePointY=separation.pointB.y,physicalDistance=separation.distance,
            grounded=mover.IsGrounded,bridgeSupport=mover.SupportedRopePlatform,attached=rope.IsAttached,support=mover.SupportedRopePlatform?mover.SupportedRopePlatform.name:"bank/none",
            walking=(bool)Field(visual,"isWalking"),idle=(bool)Field(visual,"isIdlePoseDisplayed"),aiming=(bool)Field(visual,"isAimPoseDisplayed"),sprite=renderer.sprite.name,
            camera=Camera.main.transform.position,ortho=Camera.main.orthographicSize,development=Debug.isDebugBuild,
            visualBend=(float)Field(visual,"bridgeOffsetY") };
        row.facingLeft=renderer.flipX; row.alignment=RopeBodyVisual.ReviewBridgeSurfaceAlignment;
        row.slopeOffset=(float)Field(visual,"bridgeSlopeOffsetY"); row.totalOffset=row.slopeOffset+row.visualBend;
        row.input=move; row.edgeRadius=edge.edgeRadius; row.timeScale=Time.timeScale;
        row.activeHook=rope.ActiveHookPoint?rope.ActiveHookPoint.name:"none";
        row.anchor=rope.AnchorPoint; row.jointDistance=body.GetComponent<DistanceJoint2D>().distance;
        var ropeLine=rope.GetComponent<LineRenderer>();
        row.ropeOriginError=rope.IsAttached&&ropeLine.positionCount>0?Vector2.Distance(ropeLine.GetPosition(0),visual.RopeOrigin):0;
        row.aimRootError=!rope.IsAttached&&ropeLine.positionCount>0?Vector2.Distance(ropeLine.GetPosition(0),body.position):0;
        var contacts=new ContactPoint2D[32]; int n=box.GetContacts(contacts);
        for(int i=0;i<n;i++) { var c=contacts[i]; var other=c.collider==box?c.otherCollider:c.collider; row.contacts.Add(new Contact {collider=other.name,point=c.point,normal=c.normal,separation=c.separation}); }
        return row;
    }

    }
