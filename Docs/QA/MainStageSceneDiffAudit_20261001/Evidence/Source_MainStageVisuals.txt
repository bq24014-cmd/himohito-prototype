60:         public static bool Apply(GameObject player)
61:         {
62:             bool changed = false;
63:             changed |= RecoverySwitchVisual.Ensure(
64:                 FindSceneObject(MainStageSectionThreeSetup.RecoverySwitchName));
65:             changed |= HimoHitoCraftRoomBackground.Ensure(
66:                 BackgroundName, FarBackgroundName);
67:             changed |= RestorePlayerVisual(player);
68:             changed |= GoalChestPresentation.Ensure(
69:                 FindSceneObject(MainStageSectionTenSetup.GoalMarkerName),
70:                 FindSceneObject(MainStageSectionTenSetup.GoalFloorName));
71: 
72:             foreach (GameObject candidate in
73:                      Resources.FindObjectsOfTypeAll<GameObject>())
74:             {
75:                 if (candidate == null ||
76:                     !candidate.scene.IsValid() ||
77:                     !candidate.activeInHierarchy ||
78:                     !candidate.name.StartsWith(
79:                         "Main ",
80:                         System.StringComparison.Ordinal))
81:                 {
82:                     continue;
83:                 }
84: 
85:                 if (candidate == player)
86:                 {
87:                     continue;
88:                 }
89: 
90:                 if (candidate.TryGetComponent(out RopeSpikeHazard _))
91:                 {
92:                     changed |= RemoveRepeatedBlockVisuals(candidate);
93:                     changed |= RemoveChild(
94:                         candidate,
95:                         "Orange Block Platform Visual");
96:                     changed |= RemoveChild(
97:                         candidate,
98:                         "Blue Railway Platform Visual");
99:                     changed |= ApplyColor(
100:                         candidate,
101:                         new Color(1f, 0.18f, 0.25f));
102:                     if (!candidate.TryGetComponent(
103:                             out ToySpikeVisual spikeVisual))
104:                     {
105:                         spikeVisual = candidate.AddComponent<ToySpikeVisual>();
106:                         changed = true;
107:                     }
108:                     changed |= spikeVisual.Refresh();
109:                     continue;
110:                 }
111: 
112:                 if (candidate.TryGetComponent(out HookPoint _))
113:                 {
114:                     changed |= EnsureHookVisual(candidate);
115:                     continue;
116:                 }
117: 
118:                 if (candidate.name == MainStageSectionTwoSetup.BoardName)
119:                 {
120:                     changed |= EnsureLengthWindowBeamVisual(candidate);
121:                     continue;
122:                 }
123: 
124:                 if (candidate.TryGetComponent(out BoxCollider2D _) &&
125:                     candidate.TryGetComponent(out SolidSprite _))
126:                 {
127:                     if (IsBankTerrain(candidate.name))
128:                     {
129:                         changed |= EnsureBankTerrainVisual(candidate);
130:                         continue;
131:                     }
132: 
133:                     bool isStartGround = candidate.name == "Main Start Ground";
134:                     bool isSwingPassThroughRail =
135:                         candidate.TryGetComponent(
136:                             out SwingPassThroughRailPlatform _);
137:                     bool isSolidToyBoard =
138:                         !isSwingPassThroughRail &&
139:                         (isStartGround ||
140:                          candidate.TryGetComponent(out SolidSwingSurface _));
141:                     changed |= RemoveRepeatedBlockVisuals(candidate);
142:                     changed |= RemoveChild(
143:                         candidate,
144:                         isSolidToyBoard
145:                             ? "Blue Railway Platform Visual"
146:                             : "Orange Block Platform Visual");

148:                         candidate,
149:                         isSolidToyBoard ? BlockColor : RailColor);
150:                     changed |= EnsureToyVisual(
151:                         candidate,

153:                             ? "Orange Block Platform Visual"
154:                             : "Blue Railway Platform Visual",
155:                         isSolidToyBoard ? BlockResourcePath : RailResourcePath,
156:                         1);
157:                 }
158:             }
159: 
160:             return changed;

373:                 Vector2 localScale = new Vector2(
374:                     localStripWidth /
375:                     Mathf.Max(0.01f, spriteSize.x),
376:                     terrainCollider.size.y /
377:                     Mathf.Max(0.01f, spriteSize.y));

381:                     blockSprite,
382:                     localPosition,
383:                     localScale,

392:             if (sourceRenderer.enabled)
393:             {
394:                 sourceRenderer.enabled = false;
395:                 changed = true;
396:             }
397: 
398:             return changed;
399:         }

495:         private static bool RemoveRepeatedBlockVisuals(
496:             GameObject parent,
497:             int retainedStripCount)
498:         {
499:             if (parent == null)
500:             {
501:                 return false;
502:             }
503: 
504:             bool changed = false;
505:             for (int childIndex = parent.transform.childCount - 1;
506:                  childIndex >= 0;
507:                  childIndex--)
508:             {
509:                 Transform child = parent.transform.GetChild(childIndex);
510:                 if (!child.name.StartsWith(
511:                         BlockTileVisualPrefix,
512:                         System.StringComparison.Ordinal))
513:                 {
514:                     continue;
515:                 }
516: 
517:                 string suffix = child.name.Substring(
518:                     BlockTileVisualPrefix.Length);
519:                 bool keep = int.TryParse(suffix, out int stripNumber) &&
520:                             stripNumber >= 1 &&
521:                             stripNumber <= retainedStripCount;
522:                 if (keep)
523:                 {
524:                     continue;
525:                 }
526: 
527:                 if (Application.isPlaying)
528:                 {
529:                     Object.Destroy(child.gameObject);
530:                 }
531:                 else
532:                 {
533:                     Object.DestroyImmediate(child.gameObject);
534:                 }
535:                 changed = true;
536:             }
537: 
538:             return changed;
539:         }

733:                 CraftWoodPlatformVisual.TryEnsure(target, out bool woodChanged)) return woodChanged;

735:             bool changed = false;

738:                 CraftRailPlatformVisual.TryEnsure(target, out changed)) return changed;

740:             Sprite processedSprite =
741:                 TutorialFirstSectionVisuals.LoadProcessedToySprite(resourcePath);
742:             if (processedSprite == null)
743:             {
744:                 // A Play-mode transition can destroy SolidSprite's temporary
745:                 // sprite while leaving the renderer enabled. Reapply the same
746:                 // color to rebuild a visible fallback before exposing it.
747:                 if (target.TryGetComponent(out SolidSprite fallbackVisual))

766:                 changed = true;

775:             Vector3 targetScale = new Vector3(
776:                 1f / Mathf.Max(0.01f, spriteSize.x),
777:                 1f / Mathf.Max(0.01f, spriteSize.y),
778:                 1f);
779:             Vector3 targetLocalPosition = Vector3.zero;
780:             if (target.name == MainStageSectionTenSetup.GoalMarkerName &&

784:                 // Keep the goal trigger and its gameplay position unchanged.
785:                 float floorTop = MainStageSectionTenSetup.GoalFloorPosition.y +
786:                     MainStageSectionTenSetup.GoalFloorSize.y * 0.5f;
787:                 const float woodSurfaceInset = 0.16f;
788:                 float visualWorldHeight = spriteSize.y * targetScale.y *
789:                     Mathf.Abs(target.transform.lossyScale.y);
790:                 targetLocalPosition = target.transform.InverseTransformPoint(
791:                     new Vector3(target.transform.position.x,
792:                         floorTop - woodSurfaceInset + visualWorldHeight * 0.5f,
793:                         target.transform.position.z));
794:             }
795:             if (visualTransform.localPosition != targetLocalPosition)
796:             {
797:                 visualTransform.localPosition = targetLocalPosition;
798:                 changed = true;
799:             }
800:             if (visualTransform.localRotation != Quaternion.identity)
801:             {
802:                 visualTransform.localRotation = Quaternion.identity;
803:                 changed = true;
804:             }
805:             if (visualTransform.localScale != targetScale)
806:             {
807:                 visualTransform.localScale = targetScale;
808:                 changed = true;
809:             }
810: 
811:             if (!visualTransform.TryGetComponent(out SpriteRenderer renderer))
812:             {
813:                 renderer = visualTransform.gameObject.AddComponent<SpriteRenderer>();
814:                 changed = true;
815:             }
816:             if (renderer.sprite != processedSprite)
817:             {
818:                 renderer.sprite = processedSprite;
819:                 changed = true;
820:             }
821:             if (resourcePath == RailResourcePath && renderer.drawMode != SpriteDrawMode.Simple)
822:             {
823:                 renderer.drawMode = SpriteDrawMode.Simple;
824:                 changed = true;
825:             }
826:             if (!renderer.enabled)
827:             {
828:                 renderer.enabled = true;
829:                 changed = true;
830:             }
831:             Color targetColor = tint ?? Color.white;
832:             if (renderer.color != targetColor)
833:             {
834:                 renderer.color = targetColor;
835:                 changed = true;
836:             }
837:             if (renderer.sortingLayerID != sourceRenderer.sortingLayerID)
838:             {
839:                 renderer.sortingLayerID = sourceRenderer.sortingLayerID;
840:                 changed = true;
841:             }
842: 
843:             int targetOrder = sourceRenderer.sortingOrder + sortingOrderOffset;
844:             if (renderer.sortingOrder != targetOrder)
845:             {
846:                 renderer.sortingOrder = targetOrder;
847:                 changed = true;
848:             }
849:             if (sourceRenderer.enabled)
850:             {
851:                 sourceRenderer.enabled = false;
852:                 changed = true;
853:             }
854: 
855:             if (resourcePath == BlockResourcePath)
856:                 changed |= WoodenPlatformDepthVisual.Ensure(target);
857:             return changed;
