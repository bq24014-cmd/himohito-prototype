1: using HimoHito;
2: using UnityEditor;
3: using UnityEditor.SceneManagement;
4: using UnityEngine.SceneManagement;
5: 
6: namespace HimoHitoEditor
7: {
8:     /// <summary>Keeps the saved MainStage scene aligned with its setup code.</summary>
9:     [InitializeOnLoad]
10:     public static class MainStageSceneSync
11:     {
12:         static MainStageSceneSync()
13:         {
14:             EditorSceneManager.sceneOpened += OnSceneOpened;
15:             EditorApplication.playModeStateChanged += OnPlayModeStateChanged;
16:             EditorApplication.delayCall += SyncActiveScene;
17:         }
18: 
19:         private static void OnSceneOpened(Scene scene, OpenSceneMode mode)
20:         {
21:             Sync(scene);
22:         }
23: 
24:         private static void SyncActiveScene()
25:         {
26:             Sync(SceneManager.GetActiveScene());
27:         }
28: 
29:         private static void OnPlayModeStateChanged(PlayModeStateChange state)
30:         {
31:             if (state == PlayModeStateChange.EnteredEditMode)
32:             {
33:                 EditorApplication.delayCall += SyncActiveScene;
34:             }
35:         }
36: 
37:         private static void Sync(Scene scene)
38:         {
39:             if (EditorApplication.isPlayingOrWillChangePlaymode ||
40:                 !scene.IsValid() ||
41:                 scene.name != "MainStage")
42:             {
43:                 return;
44:             }
45: 
46:             if (MainStageFloorCollisionSetup.ApplyCurrentScene())
47:             {
48:                 EditorSceneManager.MarkSceneDirty(scene);
49:                 EditorSceneManager.SaveScene(scene);
50:             }
51:         }
52:     }
53: }
